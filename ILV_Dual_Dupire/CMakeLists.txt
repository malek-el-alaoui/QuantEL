cmake_minimum_required(VERSION 3.16)
project(ILV_Dual_Dupire_CPP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dual_arbitrage_margin src/dual_arbitrage_margin.cpp)
add_executable(extended_dupire_comparison src/extended_dupire_comparison.cpp)
if (MSVC)
  target_compile_options(dual_arbitrage_margin PRIVATE /W4 /O2)
  target_compile_options(extended_dupire_comparison PRIVATE /W4 /O2)
else()
  target_compile_options(dual_arbitrage_margin PRIVATE -Wall -Wextra -O3)
  target_compile_options(extended_dupire_comparison PRIVATE -Wall -Wextra -O3)
endif()
